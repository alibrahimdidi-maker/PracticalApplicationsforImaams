تمهيد للموضوع السادس

6 ވަނަ ބައިގައި ބަލައިލާނީ، ހުކުރު ދުވަހުގެ މާތްކަމާއި، ހުކުރު ނަމާދުގެ ފިޤުހީ ބައިތަކުގެ މައްޗަށެވެ. މި ބައިގައި ބޭނުންކުރެވިފައިވާ ފިޤުހީ މަޞްދަރަކީ السيد سابق ގެ «فقه السنة» ގެ «الجمعة» ބައި (ص. 212-220) އެވެ.

فضل يوم الجمعة

فقه السنة ގައި ބަޔާންވެގެންވެއެވެ: «خَيْرُ يَوْمٍ طَلَعَتْ فِيهِ الشَّمْسُ يَوْمُ الْجُمُعَةِ، فِيهِ خُلِقَ آدَمُ، وَفِيهِ أُدْخِلَ الْجَنَّةَ، وَفِيهِ أُخْرِجَ مِنْهَا، وَلَا تَقُومُ السَّاعَةُ إِلَّا فِي يَوْمِ الْجُمُعَةِ» (އިރު އެރި ދުވަސްތަކުގެ ތެރެއިން އެންމެ ހެޔޮވެގެންވާ ދުވަހަކީ ހުކުރު ދުވަހެވެ، آدم ޢަލައިހިއްސަލާމް ހެއްދެވުނީ މި ދުވަހުއެވެ، ސުވަރުގެ ވެއްދެވުނީ މި ދުވަހުއެވެ، ސުވަރުގެއިން ނެރެވުނީވެސް މި ދުވަހުއެވެ، ޤިޔާމަތްވެސް ޤާއިމުވާނީ ހުކުރު ދުވަހުއެވެ) (رواه مسلم وأبو داود والنسائي والترمذي وصححه، ص.212).

ساعة الإجابة يوم الجمعة

«إِنَّ فِي الْجُمُعَةِ سَاعَةً لَا يُوَافِقُهَا عَبْدٌ مُسْلِمٌ يُصَلِّي يَسْأَلُ اللَّهَ فِيهَا خَيْرًا إِلَّا أَعْطَاهُ إِيَّاهُ، وَهِيَ بَعْدَ الْعَصْرِ» (رواه أحمد، ص.213). ބައެއް ބޭފުޅުން ބަޔާންކުރައްވަނީ، «مَا بَيْنَ أَنْ يَجْلِسَ الْإِمَامُ إِلَى أَنْ تُقْضَى الصَّلَاةُ» (إمام މިންބަރުމަތީ އިށީނދެވަޑައިގަތީއްސުރެ، ނަމާދު ނިމެންދެންވާ ވަގުތުކޮޅު) ކަމުގައެވެ.

سنن يوم الجمعة (2)

«التَّبْكِيرُ إِلَى الْجُمُعَةِ» (ހުކުރަށް އަވަހަށް ދިއުން) — «مَنِ اغْتَسَلَ يَوْمَ الْجُمُعَةِ غُسْلَ الْجَنَابَةِ ثُمَّ رَاحَ فِي السَّاعَةِ الْأُولَى فَكَأَنَّمَا قَرَّبَ بَدَنَةً، وَمَنْ رَاحَ فِي السَّاعَةِ الثَّانِيَةِ فَكَأَنَّمَا قَرَّبَ بَقَرَةً» (متفق عليه، ص.215) — 1 ވަނަ ގަޑިއިރުގައި ދިއުމުން ޖަމަލެއް ސަދަޤާތްކުރި ފަދަ ސަވާބު، 2 ވަނަ ގަޑިއިރުގައި ދިއުމުން ގެރިއެއް ސަދަޤާތްކުރި ފަދަ ސަވާބު ލިބެއެވެ.

حكم صلاة الجمعة

«أَجْمَعَ الْعُلَمَاءُ عَلَى أَنَّ صَلَاةَ الْجُمُعَةِ فَرْضُ عَيْنٍ، وَأَنَّهَا رَكْعَتَانِ» (ص.217). ދަލީލެއްގެ ގޮތުގައި: ﴿يَا أَيُّهَا الَّذِينَ آمَنُوا إِذَا نُودِيَ لِلصَّلَاةِ مِنْ يَوْمِ الْجُمُعَةِ فَاسْعَوْا إِلَى ذِكْرِ اللَّهِ وَذَرُوا الْبَيْعَ﴾ [الجمعة: 9].

الخلاف في عدد المنعقدين

«لَا خِلَافَ بَيْنَ الْعُلَمَاءِ فِي أَنَّ الْجَمَاعَةَ شَرْطٌ مِنْ شُرُوطِ صِحَّةِ الْجُمُعَةِ... وَاخْتَلَفُوا فِي الْعَدَدِ الَّذِي تَنْعَقِدُ بِهِ الْجُمُعَةِ إِلَى خَمْسَةَ عَشَرَ مَذْهَبًا ذَكَرَهَا الْحَافِظُ فِي الْفَتْحِ» (ص.219) — الشوكاني ބަޔާންކުރައްވަނީ، الراجح (އެންމެ ބުރަދަން ބޮޑު ރައުޔަކީ) 2 މީހުންނަށްވެސް ފުދޭ ކަމުގައެވެ.

آداب الذهاب إلى الجمعة

«تَخَطِّي الرِّقَابِ» (މީސްތަކުންގެ ކޮނޑުމައްޗާއި، ފައިމަތިން ގިރާވެގެން ސަފުގެ ކުރިއަށް ދިއުން) كره ކުރައްވައެވެ. ދަލީލެއްގެ ގޮތުގައި: «اجْلِسْ فَقَدْ آذَيْتَ وَآنَيْتَ» (رواه أبو داود والنسائي، ص.216) — ފުރަތަމަ ސަފަށް ދިއުމަށްޓަކައި، ފަހުން ދިޔަ ބައިވެރިއަކު، ބައިވެރިވި ފަރާތްތަކުގެ ދޭތެރެއިން ގިރާވެގެން ދިއުމުން ދުރުހެލިވުމަށް ބާރުއެޅިފައިވެއެވެ.

حكم الإنصات للخطبة

«مَنِ اغْتَسَلَ يَوْمَ الْجُمُعَةِ ثُمَّ أَتَى الْجُمُعَةَ فَصَلَّى مَا قُدِّرَ لَهُ، ثُمَّ أَنْصَتَ حَتَّى يَفْرُغَ مِنْ خُطْبَتِهِ، ثُمَّ يُصَلِّي مَعَهُ غُفِرَ لَهُ مَا بَيْنَهُ وَبَيْنَ الْجُمُعَةِ الْأُخْرَى وَفَضْلُ ثَلَاثَةِ أَيَّامٍ» (رواه مسلم، ص.216) — ޚުޠުބާ އަޑުއަހަމުން، ސީދާ ވާހަކަދެއްކުމުން ދުރުހެލިވުމަށް ބާރުއެޅިފައިވެއެވެ.

ما يفعل من فاتته الجمعة

ފުޤަހާއިން ބަޔާންކުރައްވާ ގޮތުގައި، ސީދާ ފުޤުހީ ސަބަބެއް ނެތި، ހުކުރު ނަމާދު ފާއިތުވެއްޖެ ފަރާތެއް، ބަދަލުގައި، 4 ރަކުޢަތުގެ ظُهر ނަމާދު ފުއްދާނެއެވެ. ދަލީލެއްގެ ގޮތުގައި: «مَنْ تَرَكَ ثَلَاثَ جُمَعٍ تَهَاوُنًا طَبَعَ اللَّهُ عَلَى قَلْبِهِ» (رواه أبو داود والترمذي وابن ماجه، ص.217).

وقت صلاة الجمعة

ފުޤަހާ ބާރުއަޅުއްވާ ގޮތުގައި، إمام އެއް، ހުކުރު ދުވަހަށްޓަކައި، ސުންނަތްތައް (غسل، سواك، طيب) ފުއްދައި، ފުރިހަމަ ފެންވަރެއްގައި ޚުޠުބާއަށް ތައްޔާރުވުން މުހިންމެވެ.

خاتمة

ހުކުރު ދުވަހަކީ، مسلم ން ގެ ދުވަސްތަކުގެ ތެރެއިން އެންމެ ހެޔޮވެގެންވާ ދުވަހެވެ. ހުކުރު ނަމާދަކީ، ފިރިހެނުންނަށް فرض عين ވެގެންވާ، ފުޅާ ފިޤުހީ ބައިތަކެއް ހިމެނޭ ނަމާދެކެވެ. ﷲ ސުބުޙާނަހޫ ވަތަޢާލާ، ފުރިހަމަ ފެންވަރެއްގައި ހުކުރު ނަމާދު ފުއްދުމުގެ ތައުފީޤު ދެއްވާށިއެވެ! آمين.
