تمهيد للموضوع الخامس

5 ވަނަ ބައިގައި ބަލައިލާނީ، ފަރުޟު 5 ނަމާދުގެ ފިޤުހީ ބައިތަކުގެ މައްޗަށެވެ — ނަމާދުގެ މާނަ، ފަރުޟުވި ގޮތް، ކޮންމެ ނަމާދެއްގެ ވަގުތު، ޝަރުޠުތައް، ރުކުންތައް، އަދި ޢަމަލީ ފެންވަރުގެ ފަރިތަކުރުންތަކުގެ މައްޗަށެވެ. މި ބައިގައި ބޭނުންކުރެވިފައިވާ ފިޤުހީ މަޞްދަރަކީ السيد سابق ގެ «فقه السنة» ގެ «مواقيت الصلاة» ބައި (ص. 67-74) އެވެ.

الصلاة ގެ ލުޣަވީ މާނައަކީ دعاء (ދުޢާ) އެވެ. ޝަރުޢީ މާނައަކީ، ސީދާ ބަސްތަކާއި ޢަމަލުތަކަކުން، تكبير ން ފެށިގެން، تسليم ން ނިމޭ، ޚާއްޞަ ފެންވަރެއްގައި ﷲއަށް ކުރެވޭ އަޅުކަމެކެވެ.

وقت صلاة الفجر

«وقت صبح من طلوع الفجر ما لم تطلع الشمس، فإذا طلعت الشمس فأمسك عن الصلاة» (رواه مسلم عن عبد الله بن عمرو) — ފަތިސް ނަމާދުގެ ވަގުތު ފެށެނީ ފަޖުރު ފެނުމުންނެވެ، ނިމެނީ އިރު އަރަންދެނެވެ.

وقت صلاة الظهر

«وقت الظهر إذا زالت الشمس، وكان ظل الرجل كطوله ما لم يحضر العصر» (رواه مسلم عن عبد الله بن عمرو) — މެންދުރު ނަމާދުގެ ވަގުތު ފެށެނީ، އިރު މެދުއުޑުން ފެއިލައިލީމައެވެ، ނިމެނީ، ކޮންމެ އެއްޗެއްގެ ހިޔަނި، އެއެއްޗެއްގެ ދިގުމިނާ އެއްވަރުވުމުންނެވެ.

وقت صلاة العصر

«ووقت العصر ما لم تصفر الشمس» (رواه مسلم) — ޢަޞްރު ނަމާދުގެ ވަގުތު ފެށެނީ، ކޮންމެ އެއްޗެއްގެ ހިޔަނި، އެއެއްޗެއްގެ ދިގުމިނާ އެއްވަރުވުމުންނެވެ، ނިމެނީ، އިރު އޮއްސެންދެނެވެ. ދަލީލެއްގެ ގޮތުގައި: «مَنْ أَدْرَكَ رَكْعَةً مِنَ الْعَصْرِ قَبْلَ أَنْ تَغْرُبَ الشَّمْسُ فَقَدْ أَدْرَكَ الْعَصْرَ» (متفق عليه عن أبي هريرة، ص.69).

وقت صلاة العشاء

«يدخل وقت صلاة العشاء بمغيب الشفق الأحمر، ويمتد إلى نصف الليل» (ص.71) — ޢިޝާ ނަމާދުގެ ވަގުތު ފެށެނީ، ރަތްކުލައިގެ شفق ގެއްލުމުންނެވެ، ނިމެނީ، ރޭގަނޑުގެ މެދާ ހަމައަށެވެ. «لَوْلَا أَنْ أَشُقَّ عَلَى أُمَّتِي لَأَمَرْتُهُمْ أَنْ يُؤَخِّرُوا الْعِشَاءَ إِلَى ثُلُثِ اللَّيْلِ أَوْ نِصْفِهِ» (رواه أحمد والترمذي وابن ماجه، حسن).

«مَنْ نَامَ عَنْ صَلَاةٍ أَوْ نَسِيَهَا فَلْيُصَلِّهَا إِذَا ذَكَرَهَا، لَا كَفَّارَةَ لَهَا إِلَّا ذَلِكَ» (رواه البخاري ومسلم، حديث قتادة عن أبي هريرة) — ނިދި ފުރި، ނުވަތަ ހަނދާން ނެތިގެން، ނަމާދެއް ފާއިތުވެއްޖެނަމަ، ހަނދާންވުމުން، ސީދާ ޤަޟާކުރުން ފަރުޟުވެގެންވެއެވެ.
